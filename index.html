canvas.addEventListener('click', () => {
  if (document.pointerLockElement !== canvas && canvas.requestPointerLock) {
    try { canvas.requestPointerLock(); } catch (err) {}
  }
});
canvas.addEventListener('contextmenu', (e) => e.preventDefault());
canvas.addEventListener('mousedown', (e) => {
  soundEngine.init();
  if (e.button === 0) { state.isFiring = true; firePrimaryWeapon(); }
  if (e.button === 2) { state.isAiming = true; }
  state.isDragging = true;
  state.lastMouseX = e.clientX;
  state.lastMouseY = e.clientY;
});
window.addEventListener('mousemove', (e) => {
  const locked = document.pointerLockElement === canvas;
  if (!locked && !state.isDragging) return;
  const dx = locked ? e.movementX : (e.clientX - state.lastMouseX);
  const dy = locked ? e.movementY : (e.clientY - state.lastMouseY);
  state.lastMouseX = e.clientX;
  state.lastMouseY = e.clientY;
  const sens = state.isAiming ? 0.0022 : 0.0038;
  state.camYaw -= dx * sens;
  state.camPitch = THREE.MathUtils.clamp(state.camPitch + dy * sens, -0.35, 1.32);
  state.camFollowTimer = 1.8;
});
window.addEventListener('mouseup', (e) => {
  if (e.button === 0) state.isFiring = false;
  if (e.button === 2) state.isAiming = false;
  state.isDragging = false;
});

⌨️ دليل تحكم GTA VI الجديد (الماوس + الدريفت + الطيران)

🖱️ تحكم الماوس والمشي والتصويب (TPS):
حركة الماوس: دوران كاميرا حر 360°
كليك يمين: زووم تصويب الكتف (GTA Aim)
كليك يسار (مستمر): إطلاق نار رشاش
W A S D + Shift: المشي والجري السريع
عجلة الماوس / 1-4: تبديل السلاح الفوري
F أو E: ركوب / نزول من السيارة أو الطائرة
🏎️ قيادة السيارات والدريفت + ✈️ الطيران:
Space (في السيارة): فرامل يد ودريفت احترافي
Shift (في السيارة/الطائرة): نيترو / حارق لاحق
W / S + A / D: تسارع وانعطاف سلس
الأسهم أو الماوس: توجيه الطائرة والميلان
M: إطلاق صاروخ موجه ذكي
X: إطلاق شعلات حرارية مضادة للصواريخ
